func (h *Handlers) Write(ctx context.Context,
    r *core.Request, a *WRITE3args) (*WRITE3res, error) {
  o := vfs.Anonymous(a.File) // NFSv3 has no opens
  w, err := h.svc.Write(ctx, r.Call, o,
    int64(a.Offset), a.Data, stability(a.Stable))
  if err != nil {
    return nil, err // mapped once, by the table
  }
  return &WRITE3res{Count: uint32(w.N),
    Committed: a.Stable, Verf: w.Verifier,
    Wcc: w.WCC}, nil
}
